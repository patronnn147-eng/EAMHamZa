### Task 2: Create embedder.py — singleton model + async-safe embed

**File:** `app/rag-service/embedder.py`

```python
"""
Embedding model singleton.

SentenceTransformer.encode() is synchronous.
Always call async wrappers (embed_text / embed_batch) — they use run_in_executor.
"""
import asyncio
import logging
from typing import List

logger = logging.getLogger(__name__)

_model = None


def _load_model():
    global _model
    if _model is None:
        logger.info("Loading all-MiniLM-L6-v2...")
        from sentence_transformers import SentenceTransformer
        _model = SentenceTransformer("sentence-transformers/all-MiniLM-L6-v2")
        logger.info("Embedding model ready.")
    return _model


def _sync_embed(text: str) -> List[float]:
    return _load_model().encode(text, normalize_embeddings=True).tolist()


def _sync_embed_batch(texts: List[str]) -> List[List[float]]:
    return [v.tolist() for v in _load_model().encode(
        texts, normalize_embeddings=True, batch_size=32
    )]


async def embed_text(text: str) -> List[float]:
    """Async-safe single embed."""
    loop = asyncio.get_event_loop()
    return await loop.run_in_executor(None, _sync_embed, text)


async def embed_batch(texts: List[str]) -> List[List[float]]:
    """Async-safe batch embed."""
    loop = asyncio.get_event_loop()
    return await loop.run_in_executor(None, _sync_embed_batch, texts)


def vec_to_str(vector: List[float]) -> str:
    """Convert float list → pgvector string: [0.123,0.456,...]"""
    return "[" + ",".join(f"{x:.8f}" for x in vector) + "]"
```

### Task 3: Create ingestor.py — text extraction + chunking + ingestion pipeline

**File:** `app/rag-service/ingestor.py`

```python
"""
Document ingestion pipeline.

Flow: file bytes → extract text (PDF/TXT) → sentence-aware chunking
      → batch embed → store Document + DocChunk rows to pgvector.
"""
import logging
import re
import uuid
from datetime import datetime
from typing import List, Optional, Tuple

from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncSession

from embedder import embed_batch, vec_to_str

logger = logging.getLogger(__name__)

CHUNK_WORDS = 512
OVERLAP_WORDS = 50
MIN_CHUNK_WORDS = 10


def extract_pdf(file_bytes: bytes) -> List[Tuple[int, str]]:
    """Extract (page_num, text) per page. import fitz — NOT import pymupdf."""
    import fitz  # pymupdf internal module name is fitz

    pages = []
    try:
        with fitz.open(stream=file_bytes, filetype="pdf") as pdf:
            for num, page in enumerate(pdf, start=1):
                t = page.get_text("text").strip()
                if t and len(t) > 20:
                    pages.append((num, t))
    except Exception as e:
        raise ValueError(f"PDF extraction failed: {e}") from e
    return pages


def chunk_text(text_: str) -> List[str]:
    """Sentence-aware chunking: split on sentence boundaries, add word overlap."""
    sentences = re.split(r'(?<=[.!?])\s+', text_)
    chunks, current, length = [], [], 0

    for sent in sentences:
        words = sent.split()
        if not words:
            continue
        if length + len(words) > CHUNK_WORDS and current:
            chunks.append(" ".join(current))
            current = (current[-OVERLAP_WORDS:] if len(current) >= OVERLAP_WORDS else current[:]) + words
            length = len(current)
        else:
            current.extend(words)
            length += len(words)

    if current and len(current) >= MIN_CHUNK_WORDS:
        chunks.append(" ".join(current))

    return [c.strip() for c in chunks if len(c.split()) >= MIN_CHUNK_WORDS]


async def ingest_document(
    file_bytes: bytes,
    filename: str,
    doc_type: str,
    db: AsyncSession,
    machine_id: Optional[int] = None,
    uploaded_by: Optional[int] = None,
    description: Optional[str] = None,
) -> dict:
    """
    Full ingestion pipeline. Returns {doc_id, chunk_count}.
    Uses raw SQL inserts for pgvector compatibility.
    """
    # 1. Extract text
    if filename.lower().endswith(".pdf"):
        pages = extract_pdf(file_bytes)
    else:
        pages = [(1, file_bytes.decode("utf-8", errors="ignore"))]

    if not pages:
        raise ValueError("No extractable text found.")

    # 2. Insert document row
    doc_id = str(uuid.uuid4())
    await db.execute(text("""
        INSERT INTO documents (id, filename, doc_type, description, machine_id, uploaded_by, file_size_bytes, created_at)
        VALUES (:id, :filename, :doc_type, :description, :machine_id, :uploaded_by, :file_size, now())
    """), {
        "id": doc_id, "filename": filename, "doc_type": doc_type,
        "description": description, "machine_id": machine_id,
        "uploaded_by": uploaded_by, "file_size": len(file_bytes),
    })

    # 3. Build chunks + metadata
    all_texts, all_meta = [], []
    for page_num, page_text in pages:
        for chunk in chunk_text(page_text):
            all_texts.append(chunk)
            all_meta.append({
                "page": page_num, "filename": filename,
                "doc_type": doc_type, "machine_id": machine_id,
            })

    if not all_texts:
        raise ValueError("Document produced no valid chunks.")

    # 4. Batch embed (async-safe)
    logger.info(f"Embedding {len(all_texts)} chunks for '{filename}'...")
    vectors = await embed_batch(all_texts)

    # 5. Insert chunks — cast string to vector(384) in SQL
    import json
    for idx, (chunk, meta, vector) in enumerate(zip(all_texts, all_meta, vectors)):
        await db.execute(text("""
            INSERT INTO doc_chunks (id, doc_id, chunk_index, content, embedding, metadata, created_at)
            VALUES (gen_random_uuid(), :doc_id, :idx, :content,
                    CAST(:embedding AS vector(384)),
                    :metadata::jsonb, now())
        """), {
            "doc_id": doc_id, "idx": idx, "content": chunk,
            "embedding": vec_to_str(vector),
            "metadata": json.dumps(meta),
        })

    # 6. Update chunk_count
    await db.execute(text("""
        UPDATE documents SET chunk_count = :count WHERE id = :id
    """), {"count": len(all_texts), "id": doc_id})

    await db.commit()
    logger.info(f"Ingested '{filename}': {len(all_texts)} chunks, machine_id={machine_id}")
    return {"doc_id": doc_id, "chunk_count": len(all_texts)}
```

### Task 4: Create retriever.py — cosine similarity search

**File:** `app/rag-service/retriever.py`

```python
"""
Semantic retriever — cosine similarity search via pgvector.
"""
import logging
from typing import Any, Dict, List, Optional

from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncSession

from embedder import embed_text, vec_to_str

logger = logging.getLogger(__name__)


async def retrieve_chunks(
    query: str,
    db: AsyncSession,
    machine_id: Optional[int] = None,
    top_k: int = 3,
    threshold: float = 0.7,
) -> List[Dict[str, Any]]:
    """
    Embed query → cosine similarity search → return top-K chunks above threshold.
    machine_id=None searches all documents.
    Returns [] on any error — caller degrades gracefully.
    """
    query_vec = await embed_text(query)
    vec_str = vec_to_str(query_vec)

    sql = text("""
        SELECT
            dc.content,
            dc.metadata,
            1 - (dc.embedding <=> CAST(:vec AS vector)) AS similarity
        FROM doc_chunks dc
        JOIN documents d ON dc.doc_id = d.id
        WHERE (
            CAST(:machine_filter AS integer) IS NULL
            OR d.machine_id = CAST(:machine_filter AS integer)
        )
          AND 1 - (dc.embedding <=> CAST(:vec AS vector)) > :threshold
        ORDER BY dc.embedding <=> CAST(:vec AS vector)
        LIMIT :top_k
    """)

    try:
        result = await db.execute(sql, {
            "vec": vec_str,
            "machine_filter": str(machine_id) if machine_id is not None else None,
            "threshold": threshold,
            "top_k": top_k,
        })
        rows = result.fetchall()
    except Exception as e:
        logger.error(f"pgvector retrieval error: {e}")
        return []

    return [
        {"content": r.content, "metadata": r.metadata or {}, "similarity": float(r.similarity)}
        for r in rows
    ]
```

```python
"""
EAM RAG Microservice — FastAPI app.

Endpoints (called by main backend via httpx):
  POST /ingest          — upload + ingest document
  POST /retrieve        — semantic chunk search
  GET  /documents       — list documents
  DELETE /documents/{id} — delete doc + chunks
  GET  /health          — healthcheck
"""
import json
import logging
import os
from typing import List, Optional
from uuid import UUID

from fastapi import Depends, FastAPI, File, Form, HTTPException, UploadFile, status
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
from sqlalchemy import select, text
from sqlalchemy.ext.asyncio import AsyncSession

from database import get_db
from ingestor import ingest_document
from retriever import retrieve_chunks

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

app = FastAPI(title="EAM RAG Service", version="1.0.0")

MAX_FILE_BYTES = 50 * 1024 * 1024  # 50MB
ALLOWED_EXT = {".pdf", ".txt"}


# ── Health ────────────────────────────────────────────────────────────────────

@app.get("/health")
async def health():
    return {"status": "ok", "service": "rag"}


# ── Ingest ────────────────────────────────────────────────────────────────────

class IngestResponse(BaseModel):
    doc_id: str
    chunk_count: int
    filename: str


@app.post("/ingest", response_model=IngestResponse, status_code=201)
async def ingest(
    file: UploadFile = File(...),
    doc_type: str = Form(...),
    machine_id: Optional[int] = Form(default=None),
    uploaded_by: Optional[int] = Form(default=None),
    description: Optional[str] = Form(default=None),
    db: AsyncSession = Depends(get_db),
):
    import os as _os
    ext = _os.path.splitext(file.filename or "")[1].lower()
    if ext not in ALLOWED_EXT:
        raise HTTPException(400, f"Unsupported file type '{ext}'. Use: {ALLOWED_EXT}")
    if doc_type not in ("manual", "sop", "report"):
        raise HTTPException(400, "doc_type must be: manual, sop, report")

    file_bytes = await file.read()
    if len(file_bytes) > MAX_FILE_BYTES:
        raise HTTPException(413, "File too large (max 50MB)")
    if not file_bytes:
        raise HTTPException(400, "Empty file")

    try:
        result = await ingest_document(
            file_bytes=file_bytes,
            filename=file.filename or "document",
            doc_type=doc_type,
            db=db,
            machine_id=machine_id,
            uploaded_by=uploaded_by,
            description=description,
        )
    except ValueError as e:
        raise HTTPException(422, str(e))
    except Exception as e:
        logger.error(f"Ingestion error: {e}")
        raise HTTPException(500, "Ingestion failed")

    return IngestResponse(doc_id=result["doc_id"], chunk_count=result["chunk_count"],
                          filename=file.filename or "document")


# ── Retrieve ──────────────────────────────────────────────────────────────────

class RetrieveRequest(BaseModel):
    query: str
    machine_id: Optional[int] = None
    top_k: int = 3
    threshold: float = 0.7


class ChunkResult(BaseModel):
    content: str
    metadata: dict
    similarity: float


class RetrieveResponse(BaseModel):
    chunks: List[ChunkResult]
    query: str


@app.post("/retrieve", response_model=RetrieveResponse)
async def retrieve(request: RetrieveRequest, db: AsyncSession = Depends(get_db)):
    chunks = await retrieve_chunks(
        query=request.query,
        db=db,
        machine_id=request.machine_id,
        top_k=request.top_k,
        threshold=request.threshold,
    )
    return RetrieveResponse(
        chunks=[ChunkResult(**c) for c in chunks],
        query=request.query,
    )


# ── Documents list / delete ───────────────────────────────────────────────────

class DocumentInfo(BaseModel):
    id: str
    filename: str
    doc_type: str
    description: Optional[str]
    machine_id: Optional[int]
    chunk_count: int
    file_size_bytes: Optional[int]


@app.get("/documents", response_model=List[DocumentInfo])
async def list_docs(
    doc_type: Optional[str] = None,
    machine_id: Optional[int] = None,
    db: AsyncSession = Depends(get_db),
):
    sql = "SELECT id, filename, doc_type, description, machine_id, chunk_count, file_size_bytes FROM documents WHERE 1=1"
    params = {}
    if doc_type:
        sql += " AND doc_type = :doc_type"
        params["doc_type"] = doc_type
    if machine_id is not None:
        sql += " AND machine_id = :machine_id"
        params["machine_id"] = machine_id
    sql += " ORDER BY created_at DESC LIMIT 100"

    result = await db.execute(text(sql), params)
    rows = result.fetchall()
    return [DocumentInfo(id=str(r.id), filename=r.filename, doc_type=r.doc_type,
                         description=r.description, machine_id=r.machine_id,
                         chunk_count=r.chunk_count, file_size_bytes=r.file_size_bytes)
            for r in rows]


@app.delete("/documents/{doc_id}", status_code=204)
async def delete_doc(doc_id: UUID, db: AsyncSession = Depends(get_db)):
    result = await db.execute(text("SELECT id FROM documents WHERE id = :id"), {"id": str(doc_id)})
    if not result.fetchone():
        raise HTTPException(404, "Document not found")
    # doc_chunks deleted by CASCADE FK
    await db.execute(text("DELETE FROM documents WHERE id = :id"), {"id": str(doc_id)})
    await db.commit()
```
