```python
"""
Embedding service — singleton SentenceTransformer model.

IMPORTANT: encode() is synchronous. Always call via run_in_executor in async context.
Model pre-downloaded in Dockerfile to avoid cold-start timeout.
"""
import asyncio
import logging
from typing import List

logger = logging.getLogger(__name__)

_model = None


def _get_model():
    """Lazy singleton — load model once, reuse forever."""
    global _model
    if _model is None:
        logger.info("Loading sentence-transformer model (all-MiniLM-L6-v2)...")
        from sentence_transformers import SentenceTransformer
        _model = SentenceTransformer("sentence-transformers/all-MiniLM-L6-v2")
        logger.info("Embedding model loaded.")
    return _model


def _embed_sync(text: str) -> List[float]:
    """Sync embed — must be called via run_in_executor in async context."""
    model = _get_model()
    vector = model.encode(text, normalize_embeddings=True)
    return vector.tolist()


def _embed_batch_sync(texts: List[str]) -> List[List[float]]:
    """Sync batch embed."""
    model = _get_model()
    vectors = model.encode(texts, normalize_embeddings=True, batch_size=32)
    return [v.tolist() for v in vectors]


async def embed_text(text: str) -> List[float]:
    """Async-safe text embedding. Non-blocking via thread pool."""
    loop = asyncio.get_event_loop()
    return await loop.run_in_executor(None, _embed_sync, text)


async def embed_batch(texts: List[str]) -> List[List[float]]:
    """Async-safe batch embedding."""
    loop = asyncio.get_event_loop()
    return await loop.run_in_executor(None, _embed_batch_sync, texts)


def vec_to_pgstr(vector: List[float]) -> str:
    """Convert float list to pgvector string format: [0.1,0.2,...]"""
    return "[" + ",".join(f"{x:.8f}" for x in vector) + "]"
```

```python
"""
Document ingestion pipeline.

Flow: file bytes → text extraction → chunking → batch embed → store to pgvector
"""
import logging
import re
from typing import List, Tuple, Optional

from sqlalchemy.ext.asyncio import AsyncSession

from models.documents import Document, DocChunk
from services.rag_embedder import embed_batch, vec_to_pgstr

logger = logging.getLogger(__name__)

CHUNK_SIZE_WORDS = 512
CHUNK_OVERLAP_WORDS = 50
MIN_CHUNK_WORDS = 10


def extract_pdf_text(file_bytes: bytes) -> List[Tuple[int, str]]:
    """
    Extract text per page from PDF bytes.
    Returns list of (page_number, page_text).
    """
    import fitz  # pymupdf — import name is fitz, NOT pymupdf

    pages = []
    try:
        with fitz.open(stream=file_bytes, filetype="pdf") as pdf:
            for page_num, page in enumerate(pdf, start=1):
                text = page.get_text("text").strip()
                if text and len(text) > 20:
                    pages.append((page_num, text))
    except Exception as e:
        logger.error(f"PDF extraction failed: {e}")
        raise ValueError(f"Cannot extract text from PDF: {e}") from e
    return pages


def chunk_text(text: str, chunk_size: int = CHUNK_SIZE_WORDS, overlap: int = CHUNK_OVERLAP_WORDS) -> List[str]:
    """
    Sentence-aware recursive chunking.
    Splits on sentence boundaries, respects chunk_size in words, adds overlap.
    """
    # Split on sentence-ending punctuation followed by whitespace
    sentences = re.split(r'(?<=[.!?])\s+', text)
    chunks: List[str] = []
    current: List[str] = []
    current_len = 0

    for sentence in sentences:
        words = sentence.split()
        if not words:
            continue

        if current_len + len(words) > chunk_size and current:
            # Flush current chunk
            chunks.append(" ".join(current))
            # Keep overlap from end of current
            overlap_words = current[-overlap:] if len(current) >= overlap else current[:]
            current = overlap_words + words
            current_len = len(current)
        else:
            current.extend(words)
            current_len += len(words)

    if current and len(current) >= MIN_CHUNK_WORDS:
        chunks.append(" ".join(current))

    return [c.strip() for c in chunks if len(c.split()) >= MIN_CHUNK_WORDS]


async def ingest_document(
    file_bytes: bytes,
    filename: str,
    doc_type: str,
    db: AsyncSession,
    machine_id: Optional[int] = None,
    uploaded_by: Optional[int] = None,
    description: Optional[str] = None,
) -> Document:
    """
    Full ingestion pipeline: extract → chunk → embed → store.

    Args:
        file_bytes: Raw file content (PDF or UTF-8 text)
        filename: Original filename (used for metadata + type detection)
        doc_type: manual | sop | report
        db: Async DB session
        machine_id: Optional FK to machines.id for filtered retrieval
        uploaded_by: Optional FK to utilisateurs.id
        description: Optional human description

    Returns:
        Created Document ORM object
    """
    # 1. Extract text
    if filename.lower().endswith(".pdf"):
        text_pages = extract_pdf_text(file_bytes)
    else:
        raw_text = file_bytes.decode("utf-8", errors="ignore")
        text_pages = [(1, raw_text)]

    if not text_pages:
        raise ValueError("No extractable text found in document.")

    # 2. Create document record
    doc = Document(
        filename=filename,
        doc_type=doc_type,
        description=description,
        machine_id=machine_id,
        uploaded_by=uploaded_by,
        file_size_bytes=len(file_bytes),
    )
    db.add(doc)
    await db.flush()  # assigns doc.id without full commit

    # 3. Build all chunks with metadata
    all_chunk_texts: List[str] = []
    all_metadata: List[dict] = []

    for page_num, page_text in text_pages:
        for chunk_text_ in chunk_text(page_text):
            all_chunk_texts.append(chunk_text_)
            all_metadata.append({
                "page": page_num,
                "filename": filename,
                "doc_type": doc_type,
                "machine_id": machine_id,
            })

    if not all_chunk_texts:
        raise ValueError("Document produced no valid chunks after processing.")

    # 4. Embed in batches (async, non-blocking)
    logger.info(f"Embedding {len(all_chunk_texts)} chunks for {filename}...")
    vectors = await embed_batch(all_chunk_texts)

    # 5. Store chunks
    for idx, (chunk_text_, meta, vector) in enumerate(zip(all_chunk_texts, all_metadata, vectors)):
        db.add(DocChunk(
            doc_id=doc.id,
            chunk_index=idx,
            content=chunk_text_,
            embedding=vec_to_pgstr(vector),
            metadata_=meta,
        ))

    # 6. Update chunk count and commit
    doc.chunk_count = len(all_chunk_texts)
    await db.commit()
    await db.refresh(doc)

    logger.info(f"Ingested '{filename}': {len(all_chunk_texts)} chunks, machine_id={machine_id}")
    return doc
```

### Task 3: Create rag_retriever.py

**File:** `app/backend/services/rag_retriever.py`

```python
"""
Semantic retriever — query pgvector for similar doc chunks.

Uses cosine distance (<=> operator) with optional machine_id filter.
"""
import logging
from typing import List, Optional, Dict, Any

from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncSession

from services.rag_embedder import embed_text, vec_to_pgstr

logger = logging.getLogger(__name__)

DEFAULT_TOP_K = 3
DEFAULT_THRESHOLD = 0.7


async def retrieve_chunks(
    query: str,
    db: AsyncSession,
    machine_id: Optional[int] = None,
    top_k: int = DEFAULT_TOP_K,
    threshold: float = DEFAULT_THRESHOLD,
) -> List[Dict[str, Any]]:
    """
    Embed query and retrieve top-K most similar document chunks.

    Args:
        query: User's natural language query
        db: Async DB session
        machine_id: Filter to chunks from docs linked to this machine (None = all docs)
        top_k: Max chunks to return
        threshold: Min cosine similarity (0-1). Chunks below this are excluded.

    Returns:
        List of dicts: {content, metadata, similarity}
    """
    # Embed query (async-safe via run_in_executor)
    query_vector = await embed_text(query)
    vec_str = vec_to_pgstr(query_vector)

    sql = text("""
        SELECT
            dc.content,
            dc.metadata,
            1 - (dc.embedding <=> CAST(:vec AS vector)) AS similarity
        FROM doc_chunks dc
        JOIN documents d ON dc.doc_id = d.id
        WHERE (
            :machine_id_filter IS NULL
            OR d.machine_id = CAST(:machine_id_filter AS integer)
        )
          AND 1 - (dc.embedding <=> CAST(:vec AS vector)) > :threshold
        ORDER BY dc.embedding <=> CAST(:vec AS vector)
        LIMIT :top_k
    """)

    try:
        result = await db.execute(sql, {
            "vec": vec_str,
            "machine_id_filter": str(machine_id) if machine_id is not None else None,
            "threshold": threshold,
            "top_k": top_k,
        })
        rows = result.fetchall()
    except Exception as e:
        logger.error(f"RAG retrieval failed: {e}")
        return []  # degrade gracefully — chat still works without RAG context

    chunks = [
        {
            "content": row.content,
            "metadata": row.metadata or {},
            "similarity": float(row.similarity),
        }
        for row in rows
    ]

    logger.info(f"Retrieved {len(chunks)} chunks for query (machine_id={machine_id}, threshold={threshold})")
    return chunks
```

```python
def build_rag_context(chunks: list) -> str:
    """
    Format retrieved RAG chunks into system prompt section.

    Args:
        chunks: List of dicts from rag_retriever.retrieve_chunks()
                Each: {content, metadata, similarity}

    Returns:
        Formatted [CONTEXTE DOCUMENTAIRE] string, empty string if no chunks
    """
    if not chunks:
        return ""

    lines = ["\n\n[CONTEXTE DOCUMENTAIRE]"]
    lines.append("Informations extraites de la documentation technique:")

    for chunk in chunks:
        meta = chunk.get("metadata", {})
        filename = meta.get("filename", "Document")
        page = meta.get("page")
        doc_type = meta.get("doc_type", "")
        similarity = chunk.get("similarity", 0)

        page_str = f" — page {page}" if page else ""
        type_str = f" [{doc_type}]" if doc_type else ""

        lines.append(f"\n--- Source: {filename}{page_str}{type_str} (pertinence: {similarity:.0%}) ---")
        # Cap chunk content to 600 chars to control token usage
        content = chunk["content"][:600]
        if len(chunk["content"]) > 600:
            content += "..."
        lines.append(content)

    lines.append("\n[FIN CONTEXTE DOCUMENTAIRE]")
    return "\n".join(lines)
```

```python
def build_full_system_prompt(
    role: str,
    user_name: str = "User",
    memories: Optional[List[Any]] = None,
    rag_chunks: Optional[List[Any]] = None,

    base = get_system_prompt(role, user_name)
    if memories:
        memory_section = build_memory_context(memories)
        if memory_section:
            base += memory_section
    if rag_chunks:
        rag_section = build_rag_context(rag_chunks)
        if rag_section:
            base += rag_section
    return base
```
