```typescript
const handleTelemetrySave = async () => {
  setTelemetrySaving(true);
  try {
    const token = localStorage.getItem('access_token');
    const res = await fetch(`/api/v1/ml/machines/${machine.machine_id}/telemetry`, {
      method: 'PATCH',
      headers: {
        'Content-Type': 'application/json',
        ...(token ? { Authorization: `Bearer ${token}` } : {}),
      },
      body: JSON.stringify({
        air_temperature: Number(telemetryValues.air_temperature),
        process_temperature: Number(telemetryValues.process_temperature),
        rotational_speed: Number(telemetryValues.rotational_speed),
        torque: Number(telemetryValues.torque),
        tool_wear: Number(telemetryValues.tool_wear),
      }),
    });
    if (res.ok) {
      setShowTelemetryForm(false);
      // Re-fetch prediction with new telemetry
      setLoading(true);
      const predToken = localStorage.getItem('access_token');
      const headers: HeadersInit = {};
      if (predToken) headers['Authorization'] = `Bearer ${predToken}`;
      const predRes = await fetch(`/api/v1/ml/machines/${machine.machine_id}/prediction`, { headers });
      if (predRes.ok) {
        const data = await predRes.json();
        setPrediction(data);
      }
      setLoading(false);
    }
  } finally {
    setTelemetrySaving(false);
  }
};
```
