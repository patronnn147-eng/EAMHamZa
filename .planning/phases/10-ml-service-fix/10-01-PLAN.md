        "air_temperature": 298.0,

        "rotational_speed": 1500,
        "torque": 40.0,

    try:
        result = MachineLearningService.predict_all(test_data)
        # Validate result has required fields
        required_fields = ["unified_health_score", "p1_failure_probability", "rul_days"]
        missing = [f for f in required_fields if f not in result]
        if missing:
            return {
                "status": "degraded",
                "message": f"Missing fields: {missing}",
                "result": result
            }
        # Check for NaN
        nan_fields = [f for f in required_fields if np.isnan(result.get(f, 0))]
        if nan_fields:
            return {
                "status": "degraded", 
                "message": f"NaN in fields: {nan_fields}",
                "result": result
            }
        return {"status": "healthy", "result": result}
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Prediction test failed: {str(e)}")
