Create ml_client.py:
```python
import os
import httpx
from typing import Dict, Optional, List
import logging

logger = logging.getLogger(__name__)

ML_SERVICE_URL = os.getenv("ML_SERVICE_URL", "http://localhost:8001")

class MLClient:
    """HTTP client for ML microservice."""
    
    def __init__(self, base_url: str = None):
        self.base_url = base_url or ML_SERVICE_URL
        self.client = httpx.AsyncClient(timeout=30.0)
    
    async def predict_all(self, telemetry: Dict) -> Optional[Dict]:
        """Get all P1-P6 predictions."""
        try:
            response = await self.client.post(
                f"{self.base_url}/api/v1/ml/predict-all",
                json=telemetry
            )
            response.raise_for_status()
            data = response.json()
            return data.get('predictions')
        except Exception as e:
            logger.error(f"ML prediction failed: {e}")
            return None
    
    async def predict(self, telemetry: Dict) -> Optional[Dict]:
        """Get failure probability."""
        try:
            response = await self.client.post(
                f"{self.base_url}/api/v1/ml/predict",
                json=telemetry
            )
            response.raise_for_status()
            return response.json().get('prediction')
        except Exception as e:
            logger.error(f"ML predict failed: {e}")
            return None
    
    async def health_check(self) -> bool:
        """Check if ML service is available."""
        try:
            response = await self.client.get(
                f"{self.base_url}/health"
            )
            return response.status_code == 200
        except:
            return False
    
    async def close(self):
        await self.client.aclose()

# Global client
ml_client = MLClient()

# Convenience function
async def get_ml_prediction(telemetry: Dict) -> Optional[Dict]:
    """Get ML prediction, return None if unavailable."""
    return await ml_client.predict_all(telemetry)
```
