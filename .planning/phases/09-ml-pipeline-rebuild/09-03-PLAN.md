from datetime import datetime
from typing import Dict, Optional, Any
import json
import os

class ModelRegistry:
    """Track ML model versions and metadata."""
    
    def __init__(self):
        self.models = {}
        self._load_from_disk()
    
    def register(self, model_name: str, version: str, metrics: Dict = None):
        """Register a model version."""
        entry = {
            "name": model_name,
            "version": version,
            "registered_at": datetime.now().isoformat(),
            "metrics": metrics or {}
        }
        self.models[f"{model_name}:{version}"] = entry
        self._save_to_disk()
        return entry
    
    def get(self, model_name: str, version: str = None) -> Optional[Dict]:
        """Get model metadata."""
        key = f"{model_name}:{version}" if version else model_name
        return self.models.get(key)
    
    def list_versions(self, model_name: str) -> list:
        """List all versions of a model."""
        return [
            v for k, v in self.models.items() 
            if k.startswith(model_name)
        ]
    
    def _save_to_disk(self):
        path = os.path.join(os.path.dirname(__file__), '..', 'model_registry.json')
        with open(path, 'w') as f:
            json.dump(self.models, f, indent=2)
    
    def _load_from_disk(self):
        path = os.path.join(os.path.dirname(__file__), '..', 'model_registry.json')
        if os.path.exists(path):
            with open(path) as f:
                self.models = json.load(f)

registry = ModelRegistry()

from typing import Dict, List, Optional
import numpy as np

class FeatureStore:
    """Extract and manage features for ML models."""
    
    @staticmethod
    def extract_5_features(telemetry: Dict) -> List[float]:
        """Extract 5 features: [air, process, rpm, torque, wear]"""
        return [
            float(telemetry.get('air_temperature', 298)),
            float(telemetry.get('process_temperature', 308)),
            int(telemetry.get('rotational_speed', 1500)),
            float(telemetry.get('torque', 40)),
            int(telemetry.get('tool_wear', 0))
        ]
    
    @staticmethod
    def extract_6_features(telemetry: Dict) -> List[float]:
        """Extract 6 features with temp_delta."""
        features_5 = FeatureStore.extract_5_features(telemetry)
        temp_delta = features_5[1] - features_5[0]
        return features_5 + [temp_delta]
    
    @staticmethod
    def validate_telemetry(telemetry: Dict) -> Dict:
        """Validate and sanitize telemetry input."""
        validated = {}
        
        # Validate ranges
        air = telemetry.get('air_temperature')
        if air and 250 <= air <= 350:
            validated['air_temperature'] = air
            
        process = telemetry.get('process_temperature')
        if process and 250 <= process <= 400:
            validated['process_temperature'] = process
            
        rpm = telemetry.get('rotational_speed')
        if rpm and 0 <= rpm <= 3000:
            validated['rotational_speed'] = rpm
            
        torque = telemetry.get('torque')
        if torque and 0 <= torque <= 100:
            validated['torque'] = torque
            
        wear = telemetry.get('tool_wear')
        if wear and 0 <= wear <= 300:
            validated['tool_wear'] = wear
        
        # Fill defaults for missing
        validated.setdefault('air_temperature', 298)
        validated.setdefault('process_temperature', 308)
        validated.setdefault('rotational_speed', 1500)
        validated.setdefault('torque', 40)
        validated.setdefault('tool_wear', 0)
        
        return validated
    
    @staticmethod
    def compute_derived(telemetry: Dict) -> Dict:
        """Compute derived features."""
        air = float(telemetry.get('air_temperature', 298))
        process = float(telemetry.get('process_temperature', 308))
        
        return {
            "temp_delta": process - air,
            "temp_ratio": process / air if air > 0 else 1.0,
            "power": (
                int(telemetry.get('rotational_speed', 1500)) * 
                float(telemetry.get('torque', 40))
            ),
            "wear_rate": (
                int(telemetry.get('tool_wear', 0)) / 
                max(1, int(telemetry.get('rotational_speed', 1500)))
            )
        }

feature_store = FeatureStore()
